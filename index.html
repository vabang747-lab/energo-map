// Находим, относится ли линия 0,4 кВ к выбранной ТП function isLineConnectedToTp(lineObj, tpObj) { if (!lineObj || !tpObj) return false; const tpName = String(tpObj.name || "").toLowerCase().trim(); const tpId = String(tpObj.id || "").toLowerCase().trim(); // Извлекаем чистый номер ТП (например, из "ТП-194" получаем "194") const tpNumMatch = tpName.match(/\d+/); const tpNum = tpNumMatch ? tpNumMatch[0] : null; const pId = String(lineObj.parent_id || "").toLowerCase(); const fedFrom = String(lineObj.fed_from || "").toLowerCase(); const lineName = String(lineObj.name || "").toLowerCase(); // 1. Прямая проверка по ID или parent_id if (pId && (pId === tpId || pId === tpName)) return true; if (fedFrom && (fedFrom === tpId || fedFrom === tpName)) return true; // 2. Проверка по совпадению номера/имени ТП в названии линии if (lineName.includes(tpName)) return true; if (tpNum && (lineName.includes("тп-" + tpNum) || lineName.includes("тп " + tpNum) || lineName.includes("тп" + tpNum))) return true; return false; } function refreshAll() { const selectedObj = selectedId ? objects.get(selectedId) : null; const isTpSelected = selectedObj && ["tp", "ktp", "transformer"].includes(getVoltageCategory(selectedObj)); DATA.forEach(o => { const l = layers.get(o.id); if (!l) return; const cat = getVoltageCategory(o); const isVisible = filters[cat] !== false; if (isVisible) { if (!map.hasLayer(l)) l.addTo(map); const s = styleForObject(o); const isSelected = selectedId === o.id; // Проверяем, является ли эта линия 0.4 кВ "родной" для выбранной ТП const isConnectedLine = isTpSelected && cat === "0.4" && isLineConnectedToTp(o, selectedObj); if (l.setStyle) { if (isSelected || isConnectedLine) { // Выделяем выбранный объект и его линии 0.4 кВ l.setStyle({ opacity: 1, color: isConnectedLine ? "#d63031" : s.color, // Подсвечиваем линии ТП красным/ярким цветом weight: (s.weight || 2) + 3 }); } else if (selectedId) { // Приглушаем остальные объекты при активном фокусе l.setStyle({ opacity: 0.25, color: s.color, weight: s.weight }); } else { // Стандартный вид l.setStyle({ opacity: 0.9, color: s.color, weight: s.weight }); } if (o.geometry && o.geometry.type === "Point" && l.setRadius) { l.setRadius(isSelected ? getNodeRadius(o) + 4 : getNodeRadius(o)); } } } else { if (map.hasLayer(l)) map.removeLayer(l); } }); }
