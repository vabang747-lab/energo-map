// Словарная связка для линий без parent_id в GeoJSON const EXPLICIT_CONNECTIONS = { // Имя или ID родителя (ПС / РП) -> массив названий/частей названий линий "пс «шиели»": [ "кызылту", "аэропорт", "райцентр", "залинейный", "шиелы от пс \"алмалы\"", "яч-13", "црп ввод-1", "уп", "л-95", "л-36", "35-кв", "вл-110 кв" ] // Сюда же сможем добавить другие ПС и РП, если понадобится: // "пс «алмалы»": ["линия1", "линия2"] }; // Проверка связи объекта (линии/фидера) с выбранным родителем (ПС/РП/ТП) function isLineConnectedToParent(childObj, parentObj) { if (!childObj || !parentObj) return false; const parentName = String(parentObj.name || "").toLowerCase().trim(); const parentId = String(parentObj.id || "").toLowerCase().trim(); const childName = String(childObj.name || "").toLowerCase().trim(); const pId = String(childObj.parent_id || "").toLowerCase().trim(); const fedFrom = String(childObj.fed_from || "").toLowerCase().trim(); // 1. Проверка по полю parent_id или fed_from if (pId && (pId === parentId || pId === parentName)) return true; if (fedFrom && (fedFrom === parentId || fedFrom === parentName)) return true; // 2. Проверка по словарю EXPLICIT_CONNECTIONS (для ПС "Шиели" и других) for (const [pKey, childKeywords] of Object.entries(EXPLICIT_CONNECTIONS)) { if (parentName.includes(pKey) || parentId.includes(pKey)) { if (childKeywords.some(keyword => childName.includes(keyword))) { return true; } } } // 3. Проверка для ТП 0.4 кВ по номеру (например, ТП-194 -> ВЛ-0.4кВ ТП-194) const tpNumMatch = parentName.match(/\d+/); const tpNum = tpNumMatch ? tpNumMatch[0] : null; if (childName.includes(parentName)) return true; if (tpNum && (childName.includes("тп-" + tpNum) || childName.includes("тп " + tpNum) || childName.includes("тп" + tpNum))) return true; return false; } function refreshAll() { const selectedObj = selectedId ? objects.get(selectedId) : null; DATA.forEach(o => { const l = layers.get(o.id); if (!l) return; const cat = getVoltageCategory(o); const isVisible = filters[cat] !== false; if (isVisible) { if (!map.hasLayer(l)) l.addTo(map); const s = styleForObject(o); const isSelected = selectedId === o.id; // Проверяем, привязана ли эта линия к выбранной ПС / РП / ТП const isConnectedLine = selectedObj && isLineConnectedToParent(o, selectedObj); if (l.setStyle) { if (isSelected || isConnectedLine) { // Подсвечиваем выбранный объект и все его линии l.setStyle({ opacity: 1, color: isConnectedLine ? "#ff3838" : s.color, // Подсветка отходящих линий ярким цветом weight: (s.weight || 2) + 3 }); } else if (selectedId) { // Приглушаем остальные объекты l.setStyle({ opacity: 0.2, color: s.color, weight: s.weight }); } else { // Обычный режим отображения l.setStyle({ opacity: 0.9, color: s.color, weight: s.weight }); } if (o.geometry && o.geometry.type === "Point" && l.setRadius) { l.setRadius(isSelected ? getNodeRadius(o) + 4 : getNodeRadius(o)); } } } else { if (map.hasLayer(l)) map.removeLayer(l); } }); }
